% --- 1. DEFINE UNBREAKABLE BOX STYLING ---
\tcbset{
	slidebox/.style={
		colframe=black,        % Border color
		colback=white,         % Box background color
		arc=2mm,               % Rounded corners
		boxrule=0.5mm,         % Thickness of border
		width=\textwidth,      % Fit perfectly to text block width
		before=\vspace{1em},   % Spacing above the slide
		after=\vspace{1em}     % Spacing below the slide
	}
}

% --- 2. SLIDE COUNTER + WRAPPER ENVIRONMENT ---
\newcounter{articleslide}
\renewcommand{\thearticleslide}{\arabic{articleslide}}
% Optional: reset per lecture/section instead of numbering continuously
% \@addtoreset{articleslide}{section}

% Lets \autoref{lbl} print "Slide 7" automatically
\providecommand{\articleslideautorefname}{Slide}
\def\articleslideautorefname{Slide}

% Convenience macro: \sref{lbl} -> "Slide 7" (hyperlinked)
\newcommand{\sref}[1]{\hyperref[#1]{Slide~\ref*{#1}}}

\NewEnviron{sframe}[1]{%
	\refstepcounter{articleslide}% sets \@currentlabel so \label works in the body
	\begin{center}
		\begin{tcolorbox}[slidebox]
			\setlength{\textwidth}{\linewidth}% columns are sized relative to the box interior
			{\large\bfseries Slide \thearticleslide: #1}\par\medskip
			\BODY
		\end{tcolorbox}
	\end{center}
}

% --- 3. REDIRECT BEAMER FRAMES TO THE SAFE ENVIRONMENT ---
\mode<article>{
	\let\frame\sframe
	\let\endframe\endsframe
}

% --- 4. RESTORE COLUMN LAYOUT IN ARTICLE MODE ---
\mode<article>{
	\newif\ifslidecolopen
	\global\slidecolopenfalse
	% close the current column's minipage, if one is open
	\def\slidecolclose{\ifslidecolopen\end{minipage}\global\slidecolopenfalse\fi}

\renewenvironment{columns}[1][]{%
	\par\medskip\noindent\global\slidecolopenfalse
}{%
	\slidecolclose
	\par\medskip
}

% \column[align]{width}  -- also what \begin{column}[align]{width} calls
	\renewcommand{\column}[2][t]{%
		\ifslidecolopen\slidecolclose\hfill\fi
		\begin{minipage}[#1]{#2}%
			\vspace{0pt}% makes [t] align the tops of the columns, not the first baselines
			\global\slidecolopentrue
			\setlength{\textwidth}{\linewidth}%
		}
		% \end{column} closes the minipage opened by \begin{column}
		\def\endcolumn{\slidecolclose\hfill}
	}
